from tkinter import *
window = Tk()
window.title("event handler sample")
window.geometry('100x100')
def myfunc(event):
  '''print the charatcer of the key pressed'''
  print(event.char)
window.bind("<Key>", myfunc)

def handle_click(event):
  print("\nthe button was clicked!")
button = Button(window, text="click me")
button.pack()
button.bind("<Button-1>", handle_click)
window.mainloop
